import 'package:file_picker/file_picker.dart';
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../models/site_settings.dart';
import '../../providers.dart';
import '../../widgets/admin_scaffold.dart';

class AdminSettingsScreen extends ConsumerStatefulWidget {
  const AdminSettingsScreen({super.key});

  @override
  ConsumerState<AdminSettingsScreen> createState() => _AdminSettingsScreenState();
}

class _AdminSettingsScreenState extends ConsumerState<AdminSettingsScreen> {
  final _formKey = GlobalKey<FormState>();
  late final TextEditingController _name;
  late final TextEditingController _role;
  late final TextEditingController _heroTitle;
  late final TextEditingController _heroSubtitle;
  late final TextEditingController _about;
  late final TextEditingController _years;
  late final TextEditingController _projectsShipped;
  late final TextEditingController _email;
  late final TextEditingController _phone;
  late final TextEditingController _location;
  late final Map<String, TextEditingController> _socialControllers;
  bool _available = true;
  bool _initialized = false;
  bool _saving = false;
  bool _uploadingCv = false;
  bool _uploadingProfile = false;

  @override
  void initState() {
    super.initState();
    _name = TextEditingController();
    _role = TextEditingController();
    _heroTitle = TextEditingController();
    _heroSubtitle = TextEditingController();
    _about = TextEditingController();
    _years = TextEditingController();
    _projectsShipped = TextEditingController();
    _email = TextEditingController();
    _phone = TextEditingController();
    _location = TextEditingController();
    _socialControllers = {
      for (final key in SiteSettings.defaults.socials.keys) key: TextEditingController(),
    };
  }

  @override
  void dispose() {
    for (final controller in [
      _name,
      _role,
      _heroTitle,
      _heroSubtitle,
      _about,
      _years,
      _projectsShipped,
      _email,
      _phone,
      _location,
    ]) {
      controller.dispose();
    }
    for (final controller in _socialControllers.values) {
      controller.dispose();
    }
    super.dispose();
  }

  void _load(SiteSettings settings) {
    if (_initialized) {
      return;
    }
    _initialized = true;
    _name.text = settings.name;
    _role.text = settings.role;
    _heroTitle.text = settings.heroTitle;
    _heroSubtitle.text = settings.heroSubtitle;
    _about.text = settings.about;
    _years.text = '${settings.yearsExperience}';
    _projectsShipped.text = '${settings.projectsShipped}';
    _email.text = settings.email;
    _phone.text = settings.phone;
    _location.text = settings.location;
    _available = settings.availableForHire;
    for (final entry in _socialControllers.entries) {
      entry.value.text = settings.socials[entry.key] ?? '';
    }
  }

  SiteSettings _currentSettings(SiteSettings existing) {
    return existing.copyWith(
      name: _name.text.trim(),
      role: _role.text.trim(),
      heroTitle: _heroTitle.text.trim(),
      heroSubtitle: _heroSubtitle.text.trim(),
      about: _about.text.trim(),
      yearsExperience: int.tryParse(_years.text.trim()) ?? existing.yearsExperience,
      projectsShipped: int.tryParse(_projectsShipped.text.trim()) ?? existing.projectsShipped,
      availableForHire: _available,
      email: _email.text.trim(),
      phone: _phone.text.trim(),
      location: _location.text.trim(),
      socials: {
        for (final entry in _socialControllers.entries) entry.key: entry.value.text.trim(),
      },
    );
  }

  Future<void> _save(SiteSettings existing) async {
    if (!_formKey.currentState!.validate()) {
      return;
    }
    setState(() => _saving = true);
    try {
      final updated = _currentSettings(existing);
      await ref.read(firebaseServiceProvider).saveSettings(updated);

      // Refresh the provider and repopulate local controllers with the
      // committed Firestore document so the admin UI and public UI use the
      // same source of truth.
      ref.invalidate(siteSettingsProvider);
      _initialized = false;

      if (!mounted) {
        return;
      }

      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text(
            'Settings saved successfully. Public content is synced from Firestore.',
          ),
        ),
      );
    } catch (error) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Save failed: $error')),
        );
      }
    } finally {
      if (mounted) {
        setState(() => _saving = false);
      }
    }
  }

  Future<void> _uploadCv(SiteSettings existing) async {
    final file = await FilePicker.pickFile(
      type: FileType.custom,
      allowedExtensions: const ['pdf'],
    );
    if (file == null) {
      return;
    }
    final bytes = await file.readAsBytes();
    setState(() => _uploadingCv = true);
    try {
      final url = await ref.read(firebaseServiceProvider).uploadCv(
            bytes: bytes,
            fileName: file.name,
          );
      await ref.read(firebaseServiceProvider).saveSettings(existing.copyWith(cvUrl: url));
      ref.invalidate(siteSettingsProvider);
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text(
              'CV uploaded and connected to the public Download CV button.',
            ),
          ),
        );
      }
    } catch (error) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('CV upload failed: $error')),
        );
      }
    } finally {
      if (mounted) {
        setState(() => _uploadingCv = false);
      }
    }
  }

  Future<void> _uploadProfile(SiteSettings existing) async {
    final file = await FilePicker.pickFile(type: FileType.image);
    if (file == null) {
      return;
    }
    final bytes = await file.readAsBytes();
    setState(() => _uploadingProfile = true);
    try {
      final url = await ref.read(firebaseServiceProvider).uploadProfilePhoto(
            bytes: bytes,
            fileName: file.name,
          );
      await ref.read(firebaseServiceProvider).saveSettings(existing.copyWith(profilePhotoUrl: url));
      ref.invalidate(siteSettingsProvider);
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('Profile photo uploaded.')),
        );
      }
    } catch (error) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Profile upload failed: $error')),
        );
      }
    } finally {
      if (mounted) {
        setState(() => _uploadingProfile = false);
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    final settingsAsync = ref.watch(siteSettingsProvider);
    return AdminScaffold(
      title: 'Settings',
      selectedPath: '/admin/settings',
      child: settingsAsync.when(
        loading: () => const Center(child: CircularProgressIndicator()),
        error: (_, _) => const Center(child: Text('Settings could not be loaded.')),
        data: (settings) {
          _load(settings);
          return SingleChildScrollView(
            padding: const EdgeInsets.all(24),
            child: Form(
              key: _formKey,
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text('Site settings', style: Theme.of(context).textTheme.headlineMedium),
                  const SizedBox(height: 6),
                  const Text('Everything below controls the public portfolio. Keep the experience and project counts factual.'),
                  const SizedBox(height: 26),
                  _SectionCard(
                    title: 'Identity & hero',
                    child: Column(
                      children: [
                        _twoColumns(
                          context,
                          TextFormField(controller: _name, validator: _required, decoration: const InputDecoration(labelText: 'Name')),
                          TextFormField(controller: _role, validator: _required, decoration: const InputDecoration(labelText: 'Role')),
                        ),
                        const SizedBox(height: 12),
                        TextFormField(controller: _heroTitle, validator: _required, maxLines: 2, decoration: const InputDecoration(labelText: 'Hero title')),
                        const SizedBox(height: 12),
                        TextFormField(controller: _heroSubtitle, validator: _required, maxLines: 3, decoration: const InputDecoration(labelText: 'Hero subtitle')),
                        const SizedBox(height: 12),
                        TextFormField(controller: _about, validator: _required, maxLines: 6, decoration: const InputDecoration(labelText: 'About text')),
                      ],
                    ),
                  ),
                  const SizedBox(height: 16),
                  _SectionCard(
                    title: 'Stats & availability',
                    child: Column(
                      children: [
                        _twoColumns(
                          context,
                          TextFormField(controller: _years, validator: _nonNegativeInt, keyboardType: TextInputType.number, decoration: const InputDecoration(labelText: 'Years experience')),
                          TextFormField(controller: _projectsShipped, validator: _nonNegativeInt, keyboardType: TextInputType.number, decoration: const InputDecoration(labelText: 'Projects shipped')),
                        ),
                        SwitchListTile.adaptive(
                          contentPadding: EdgeInsets.zero,
                          value: _available,
                          onChanged: (value) => setState(() => _available = value),
                          title: const Text('Available for hire'),
                          subtitle: const Text('Controls the availability badge in the hero section.'),
                        ),
                      ],
                    ),
                  ),
                  const SizedBox(height: 16),
                  _SectionCard(
                    title: 'Profile photo & CV',
                    child: Wrap(
                      spacing: 12,
                      runSpacing: 12,
                      children: [
                        FilledButton.tonalIcon(
                          onPressed: _uploadingProfile ? null : () => _uploadProfile(settings),
                          icon: _uploadingProfile ? const SizedBox(width: 18, height: 18, child: CircularProgressIndicator(strokeWidth: 2)) : const Icon(Icons.person_outline),
                          label: Text(_uploadingProfile ? 'Uploading…' : 'Upload profile photo'),
                        ),
                        FilledButton.tonalIcon(
                          onPressed: _uploadingCv ? null : () => _uploadCv(settings),
                          icon: _uploadingCv ? const SizedBox(width: 18, height: 18, child: CircularProgressIndicator(strokeWidth: 2)) : const Icon(Icons.picture_as_pdf_outlined),
                          label: Text(_uploadingCv ? 'Uploading…' : 'Upload CV PDF'),
                        ),
                        if (settings.cvUrl.isNotEmpty)
                          const Chip(avatar: Icon(Icons.check_circle_outline), label: Text('CV connected')),
                        if (settings.profilePhotoUrl.isNotEmpty)
                          const Chip(avatar: Icon(Icons.check_circle_outline), label: Text('Profile connected')),
                      ],
                    ),
                  ),
                  const SizedBox(height: 16),
                  _SectionCard(
                    title: 'Contact information',
                    child: Column(
                      children: [
                        _twoColumns(
                          context,
                          TextFormField(controller: _email, validator: _required, keyboardType: TextInputType.emailAddress, decoration: const InputDecoration(labelText: 'Email')),
                          TextFormField(controller: _phone, validator: _required, decoration: const InputDecoration(labelText: 'Phone')),
                        ),
                        const SizedBox(height: 12),
                        TextFormField(controller: _location, validator: _required, decoration: const InputDecoration(labelText: 'Location')),
                      ],
                    ),
                  ),
                  const SizedBox(height: 16),
                  _SectionCard(
                    title: 'Social links',
                    child: Column(
                      children: [
                        for (final entry in _socialControllers.entries) ...[
                          TextFormField(controller: entry.value, keyboardType: TextInputType.url, decoration: InputDecoration(labelText: _label(entry.key), prefixIcon: const Icon(Icons.link_outlined))),
                          const SizedBox(height: 12),
                        ],
                      ],
                    ),
                  ),
                  const SizedBox(height: 22),
                  Align(
                    alignment: Alignment.centerRight,
                    child: FilledButton.icon(
                      onPressed: _saving ? null : () => _save(settings),
                      icon: _saving ? const SizedBox(width: 18, height: 18, child: CircularProgressIndicator(strokeWidth: 2)) : const Icon(Icons.save_outlined),
                      label: Text(_saving ? 'Saving…' : 'Save all settings'),
                    ),
                  ),
                ],
              ),
            ),
          );
        },
      ),
    );
  }

  Widget _twoColumns(BuildContext context, Widget first, Widget second) {
    final compact = MediaQuery.sizeOf(context).width < 720;
    if (compact) {
      return Column(
        children: [
          first,
          const SizedBox(height: 12),
          second,
        ],
      );
    }
    return Row(children: [Expanded(child: first), const SizedBox(width: 12), Expanded(child: second)]);
  }

  static String _label(String value) => '${value[0].toUpperCase()}${value.substring(1)}';
  static String? _required(String? value) => value == null || value.trim().isEmpty ? 'Required' : null;
  static String? _nonNegativeInt(String? value) {
    final parsed = int.tryParse(value?.trim() ?? '');
    return parsed == null || parsed < 0 ? 'Enter a non-negative whole number' : null;
  }
}

class _SectionCard extends StatelessWidget {
  const _SectionCard({required this.title, required this.child});
  final String title;
  final Widget child;

  @override
  Widget build(BuildContext context) {
    return Card(
      child: Padding(
        padding: const EdgeInsets.all(20),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(title, style: Theme.of(context).textTheme.titleLarge),
            const SizedBox(height: 18),
            child,
          ],
        ),
      ),
    );
  }
}
